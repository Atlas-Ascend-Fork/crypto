cmake_minimum_required(VERSION 3.16)
option(CANN_BUILD_DEVICE "Build the native AICE shared library" OFF)
option(CANN_BUILD_TESTS "Build AICE tests" ON)
option(CANN_BUILD_BENCHMARKS "Build AICE benchmarks" ON)
option(CANN_BUILD_CPU_DEBUG_TESTS "Build tikicpulib CPU Kernel UT (not implemented)" OFF)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN SDK root")
set(ASCEND_COMPUTE_UNIT "ascend910b" CACHE STRING "AICE target SoC")
set(CANN_DEVICE_ID 0 CACHE STRING "ACL logical device for ST")
if(CANN_BUILD_CPU_DEBUG_TESTS)
    message(FATAL_ERROR "AICE has no tikicpulib CPU Kernel UT; use --st for device tests")
endif()
if(CANN_BUILD_DEVICE)
    if(ASCEND_COMPUTE_UNIT STREQUAL "ascend910b")
        set(AICE_ARCH dav-2201)
    elseif(ASCEND_COMPUTE_UNIT STREQUAL "ascend950")
        set(AICE_ARCH dav-3510)
    else()
        message(FATAL_ERROR "AICE supports ascend910b or ascend950")
    endif()
    list(PREPEND CMAKE_PREFIX_PATH "${ASCEND_CANN_PACKAGE_PATH}")
    set(CMAKE_ASC_ARCHITECTURES "${AICE_ARCH}" CACHE STRING "ASC architecture")
    find_package(ASC REQUIRED)
endif()
project(cann_crypto_aice LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(AICE_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)

if(CANN_BUILD_DEVICE)
    enable_language(ASC)
    add_library(cann_crypto_aice SHARED op_host/aice_host.cpp op_kernel/aice.asc)
    set_target_properties(cann_crypto_aice PROPERTIES POSITION_INDEPENDENT_CODE ON LINKER_LANGUAGE ASC)
    target_include_directories(cann_crypto_aice PUBLIC
        "$<BUILD_INTERFACE:${AICE_REPOSITORY_ROOT}/include>" "$<INSTALL_INTERFACE:include>")
    target_include_directories(cann_crypto_aice PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
    target_compile_options(cann_crypto_aice PRIVATE $<$<COMPILE_LANGUAGE:ASC>:--npu-arch=${AICE_ARCH}>)
    target_link_directories(cann_crypto_aice PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
    target_link_libraries(cann_crypto_aice PRIVATE ascendcl)

    add_executable(aice_example examples/aice_example.cpp)
    target_include_directories(aice_example PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
    target_link_directories(aice_example PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
    target_link_libraries(aice_example PRIVATE cann_crypto_aice ascendcl)

    install(TARGETS cann_crypto_aice LIBRARY DESTINATION lib)
    install(FILES "${AICE_REPOSITORY_ROOT}/include/cann_crypto/aice.h" DESTINATION include/cann_crypto)
endif()
if(CANN_BUILD_TESTS OR (CANN_BUILD_DEVICE AND CANN_BUILD_BENCHMARKS))
    add_subdirectory(tests/assets)
endif()
if(CANN_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(CANN_BUILD_BENCHMARKS)
    add_subdirectory(bench)
endif()
